  match x with
  | ((y as z)[@a ]) -> 1
  | (({ y } as z)[@a ][@b ]) -> 2
  | ((((y)[@b ]) as z)[@a ]) -> 3
  | ((y)[@b ]) as z -> 4
  | Some ((y as z)[@a ]) -> 5
  | _ -> 6
