@unboxed
type color =
  | @as("primary") Primary
  | @as("secondary") Secondary
  | Color(string)

let colorName = value =>
  switch value {
  | Color(name) => name
  | _ => "not Color"
  }

let folded = colorName(Color("primary"))

@unboxed
type number = | @as(1) One | Number(int)

let numberName = value =>
  switch value {
  | One => "one"
  | Number(_) => "number"
  }

let foldedNumber = numberName(Number(1))
